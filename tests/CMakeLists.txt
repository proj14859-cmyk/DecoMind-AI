cmake_minimum_required(VERSION 3.21)
project(DecoMindCoreTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
FetchContent_Declare(result GIT_REPOSITORY https://github.com/geode-sdk/result.git GIT_TAG v1.5.0 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(result)
FetchContent_Declare(json GIT_REPOSITORY https://github.com/geode-sdk/json.git GIT_TAG v4.0.1 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(json)
enable_testing()
add_executable(core-tests unit/core.cpp)
target_link_libraries(core-tests PRIVATE mat-json)
target_compile_options(core-tests PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall;-Wextra;-Werror;-Wno-unused-parameter>)
add_test(NAME operations-and-jsonc COMMAND core-tests ${CMAKE_CURRENT_SOURCE_DIR}/../resources/examples/patterns.jsonc)
